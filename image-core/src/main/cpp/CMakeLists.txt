cmake_minimum_required(VERSION 3.22)
project(nijiframe_pixels LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(ANDROID)
    add_library(nijiframe_pixels SHARED tone.cpp soften.cpp bridge.cpp)
    target_link_options(nijiframe_pixels PRIVATE "-Wl,-z,max-page-size=16384")
else()
    enable_testing()
    add_executable(tone_test tone.cpp soften.cpp tone_test.cpp)
    target_compile_options(tone_test PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(tone_test PRIVATE -fsanitize=address,undefined)
    add_test(NAME tone_invariants COMMAND tone_test)
endif()
